// tableExporter.js
// Vue 2.7 兼容的表格导出工具，支持嵌套表格、合并单元格、强制居中对齐和样式提取

/**
 * 表格导出工具类
 * 支持：
 * 1. 强制居中对齐
 * 2. 字体加粗和颜色提取
 * 3. 嵌套表格和合并单元格
 * 4. Vue 2.7 组件内使用或全局注册
 */

// 导入xlsx库
// 注意: 确保项目中已安装xlsx库: npm install xlsx --save
import * as XLSX from 'xlsx';

/**
 * 表格导出工具类
 */
export default class TableExporter {
  /**
   * 构造函数
   * @param {Object} options - 导出选项
   */
  constructor(options = {}) {
    this.options = {
      // 样式选项
      headerStyle: true,       // 是否应用表头样式
      zebra: false,            // 是否应用斑马纹
      borders: true,           // 是否应用边框
      wrapText: true,          // 是否启用自动换行
      defaultAlignment: 'center', // 默认文本对齐方式，强制居中
      defaultFontBold: false,  // 默认字体是否加粗
      defaultFontSize: 11,     // 默认字体大小（磅）
      defaultFontColor: '000000', // 默认字体颜色（黑色）
      // 其他选项
      fileName: '导出表格',    // 默认文件名
      debug: false,            // 是否输出调试信息
      ...options
    };
  }

  /**
   * 导出DOM中的表格为Excel
   * @param {HTMLElement|String} container - 包含表格的DOM元素或选择器
   * @param {String} fileName - 导出的文件名
   * @returns {Promise} 导出结果的Promise
   */
  exportTableFromDOM(container, fileName = this.options.fileName) {
    return new Promise((resolve, reject) => {
      try {
        // 如果传入的是选择器，获取对应的元素
        const containerElement = typeof container === 'string'
          ? document.querySelector(container)
          : container;

        if (!containerElement) {
          return reject(new Error('无法找到指定的容器元素'));
        }

        // 处理表格数据
        const processedData = this._processTableData(containerElement);

        // 导出为Excel
        return this._exportToExcel(
          processedData.data,
          processedData.merges,
          processedData.styles,
          fileName
        ).then(resolve).catch(reject);
      } catch (error) {
        reject(error);
      }
    });
  }

  /**
   * 导出数据为Excel
   * @param {Array} data - 二维数组数据
   * @param {String} fileName - 导出的文件名
   * @param {Object} sheetOptions - 工作表选项，如列宽、行高等
   * @returns {Promise} 导出结果的Promise
   */
  exportDataToExcel(data, fileName = this.options.fileName, sheetOptions = {}) {
    return new Promise((resolve, reject) => {
      try {
        // 确保data是二维数组
        if (!Array.isArray(data) || !Array.isArray(data[0])) {
          return reject(new Error('数据必须是二维数组'));
        }

        // 导出为Excel
        return this._exportToExcel(data, [], [], fileName, sheetOptions)
          .then(resolve).catch(reject);
      } catch (error) {
        reject(error);
      }
    });
  }

  /**
   * 将Vue组件中的表格导出为Excel
   * @param {Vue} vm - Vue组件实例
   * @param {String} tableSelector - 表格的CSS选择器
   * @param {String} fileName - 导出的文件名
   * @returns {Promise} 导出结果的Promise
   */
  exportVueTable(vm, tableSelector, fileName = this.options.fileName) {
    return new Promise((resolve, reject) => {
      try {
        if (!vm || !vm.$el) {
          return reject(new Error('无效的Vue组件实例'));
        }
        
        // 通过Vue实例获取表格DOM元素
        const tableElement = vm.$el.querySelector(tableSelector);
        
        if (!tableElement) {
          return reject(new Error(`未找到匹配选择器 "${tableSelector}" 的表格元素`));
        }

        // 使用DOM导出方法
        return this.exportTableFromDOM(tableElement, fileName)
          .then(resolve).catch(reject);
      } catch (error) {
        reject(error);
      }
    });
  }

  /**
   * 处理表格数据
   * @private
   * @param {HTMLElement} container - 包含表格的容器元素
   * @returns {Object} 处理后的数据
   */
  _processTableData(container) {
    const tables = container.querySelectorAll('table');
    
    if (this.options.debug) {
      console.log(`找到 ${tables.length} 个表格`);
    }
    
    // 如果只有一个表格，直接处理它
    if (tables.length === 1) {
      return this._processTableWithCorrectStructure(tables[0]);
    }
    
    // 处理多个表格的情况，假设第一个是标题，第二个是数据
    const headerTable = tables[0];
    const dataTable = tables[1];
    
    // 分析表格获取正确的列数
    const maxColumns = this._determineMaxColumns([headerTable, dataTable]);
    
    if (this.options.debug) {
      console.log(`检测到最大列数: ${maxColumns}`);
    }
    
    // 从标题表格提取标题数据
    const headerData = this._extractTableData(headerTable, maxColumns);
    
    // 从数据表格提取数据
    const bodyData = this._extractTableData(dataTable, maxColumns);
    
    // 合并标题和数据
    const combinedData = [...headerData.rows, ...bodyData.rows];
    
    // 合并合并单元格信息，注意调整行索引
    const merges = [
      ...headerData.merges,
      ...bodyData.merges.map(merge => ({
        s: { r: merge.s.r + headerData.rows.length, c: merge.s.c },
        e: { r: merge.e.r + headerData.rows.length, c: merge.e.c }
      }))
    ];
    
    return { 
      data: combinedData, 
      merges,
      styles: [...headerData.styles, ...bodyData.styles.map(style => ({
        ...style,
        row: style.row + headerData.rows.length
      }))]
    };
  }

  /**
   * 确定表格的最大列数
   * @private
   * @param {Array<HTMLTableElement>} tables - 表格元素数组
   * @returns {number} 最大列数
   */
  _determineMaxColumns(tables) {
    let maxCols = 0;
    
    tables.forEach(table => {
      const rows = table.querySelectorAll('tr');
      rows.forEach(row => {
        let colCount = 0;
        const cells = row.querySelectorAll('th, td');
        cells.forEach(cell => {
          colCount += parseInt(cell.getAttribute('colspan') || 1, 10);
        });
        maxCols = Math.max(maxCols, colCount);
      });
    });
    
    return maxCols;
  }

  /**
   * 处理单个表格的数据
   * @private
   * @param {HTMLTableElement} table - 表格元素
   * @returns {Object} 处理后的数据和合并单元格信息
   */
  _processTableWithCorrectStructure(table) {
    // 确定最大列数
    const maxColumns = this._determineMaxColumns([table]);
    
    // 提取表格数据，保持正确的列对齐
    const processedData = this._extractTableData(table, maxColumns);
    
    return { 
      data: processedData.rows, 
      merges: processedData.merges,
      styles: processedData.styles
    };
  }

  /**
   * 从表格提取数据
   * @private
   * @param {HTMLTableElement} table - 表格元素
   * @param {number} maxColumns - 最大列数
   * @returns {Object} 处理后的行数据、合并单元格信息和样式信息
   */
  _extractTableData(table, maxColumns) {
    const rows = table.querySelectorAll('tr');
    const resultRows = [];
    const merges = [];
    const styles = []; // 样式信息
    
    // 创建跟踪矩阵，记录哪些单元格被跨行单元格占用
    const spanMatrix = Array(rows.length).fill().map(() => Array(maxColumns).fill(false));
    
    // 第一遍：收集所有单元格信息
    const spanInfo = [];
    let rowIndex = 0;
    rows.forEach(row => {
      let colIndex = 0;
      const cells = row.querySelectorAll('th, td');
      
      cells.forEach(cell => {
        // 跳过被跨行单元格占用的位置
        while (colIndex < maxColumns && spanMatrix[rowIndex][colIndex]) {
          colIndex++;
        }
        
        if (colIndex >= maxColumns) return; // 超出最大列数，忽略此单元格
        
        const rowSpan = parseInt(cell.getAttribute('rowspan') || 1, 10);
        const colSpan = parseInt(cell.getAttribute('colspan') || 1, 10);
        
        // 提取单元格样式信息
        const styleInfo = this._extractCellStyle(cell);
        
        // 记录此单元格跨行跨列的信息
        spanInfo.push({
          row: rowIndex,
          col: colIndex,
          rowSpan,
          colSpan,
          value: cell.textContent.trim(),
          style: styleInfo
        });
        
        // 在占用矩阵中标记被此单元格占用的位置
        for (let r = 0; r < rowSpan; r++) {
          for (let c = 0; c < colSpan; c++) {
            if (rowIndex + r < rows.length && colIndex + c < maxColumns) {
              spanMatrix[rowIndex + r][colIndex + c] = true;
            }
          }
        }
        
        colIndex += colSpan;
      });
      
      rowIndex++;
    });
    
    // 第二遍：构建结果行，正确处理合并单元格
    // 初始化空结果数组
    for (let i = 0; i < rows.length; i++) {
      resultRows.push(Array(maxColumns).fill({ value: '' }));
    }
    
    // 填充单元格值，包括合并单元格的值
    spanInfo.forEach(info => {
      // 添加单元格值到结果数组
      resultRows[info.row][info.col] = { 
        value: info.value,
        style: info.style
      };
      
      // 保存样式信息
      styles.push({
        row: info.row,
        col: info.col,
        style: info.style
      });
      
      // 记录合并单元格信息
      if (info.rowSpan > 1 || info.colSpan > 1) {
        merges.push({
          s: { r: info.row, c: info.col },
          e: { r: info.row + info.rowSpan - 1, c: info.col + info.colSpan - 1 }
        });
        
        if (this.options.debug) {
          console.log(`检测到合并单元格: (${info.row},${info.col}) 跨 ${info.rowSpan} 行 ${info.colSpan} 列, 值: "${info.value}"`);
        }
      }
    });
    
    return { rows: resultRows, merges, styles };
  }

  /**
   * 提取单元格的样式信息
   * @private
   * @param {HTMLElement} cell - 单元格元素
   * @returns {Object} 样式信息对象
   */
  _extractCellStyle(cell) {
    // 默认样式 - 强制居中对齐
    const style = {
      alignment: {
        horizontal: 'center', // 强制水平居中
        vertical: 'center',   // 强制垂直居中
        wrapText: true        // 强制自动换行
      },
      font: {
        name: 'Arial',  // 默认字体
        sz: 11,         // 默认字号（点）
        color: { rgb: '000000' } // 默认黑色
      },
      border: { // 默认边框
        top: { style: 'thin', color: { rgb: '000000' } },
        bottom: { style: 'thin', color: { rgb: '000000' } },
        left: { style: 'thin', color: { rgb: '000000' } },
        right: { style: 'thin', color: { rgb: '000000' } }
      }
    };
    
    try {
      // 获取计算后的样式
      const computedStyle = window.getComputedStyle(cell);
      
      // 解析HTML标签中的样式
      this._parseHtmlStyles(cell, style);
      
      // 提取字体粗细
      const fontWeight = cell.style.fontWeight || computedStyle.fontWeight;
      if (fontWeight === 'bold' || parseInt(fontWeight, 10) >= 600) {
        style.font.bold = true;
      }
      
      // 提取字体大小
      const fontSize = cell.style.fontSize || computedStyle.fontSize;
      if (fontSize && fontSize !== '') {
        // 提取数字并转换为磅（约0.75倍）
        const match = fontSize.match(/(\d+(\.\d+)?)(px|pt|em|rem)?/);
        if (match) {
          let size = parseFloat(match[1]);
          const unit = match[3] || 'px';
          
          // 根据单位转换为磅
          switch (unit) {
            case 'px':
              size = size * 0.75; // 像素到磅的大致转换
              break;
            case 'pt':
              // 点已经是磅单位，不需要转换
              break;
            case 'em':
            case 'rem':
              size = size * 12; // 假设1em=12pt
              break;
          }
          
          style.font.sz = Math.round(size);
        }
      }
      
      // 提取文本颜色
      const color = cell.style.color || computedStyle.color;
      if (color && color !== '' && 
          color !== 'rgba(0, 0, 0, 0)' && 
          color !== 'rgb(0, 0, 0)') {
        style.font.color = { rgb: this._convertColorToHex(color) };
      }
      
      // 提取背景颜色
      const backgroundColor = cell.style.backgroundColor || computedStyle.backgroundColor;
      if (backgroundColor && 
          backgroundColor !== 'transparent' && 
          backgroundColor !== 'rgba(0, 0, 0, 0)' &&
          backgroundColor !== 'rgb(0, 0, 0, 0)') {
        style.fill = {
          patternType: 'solid',
          fgColor: { rgb: this._convertColorToHex(backgroundColor) }
        };
      }
      
      // 提取边框样式
      this._extractBorderStyles(cell, style, computedStyle);
      
    } catch (error) {
      if (this.options.debug) {
        console.warn('提取单元格样式时出错:', error);
      }
    }
    
    return style;
  }
  
  /**
   * 解析HTML标签中的样式
   * @private
   * @param {HTMLElement} cell - 单元格元素
   * @param {Object} style - 样式对象
   */
  _parseHtmlStyles(cell, style) {
    // 检查是否为表头单元格
    if (cell.tagName.toLowerCase() === 'th') {
      style.font.bold = true;
    }
    
    // 检查是否有<b>或<strong>标签的子元素 - 添加粗体
    if (cell.querySelector('b, strong') || cell.innerHTML.includes('<b>') || cell.innerHTML.includes('<strong>')) {
      style.font.bold = true;
    }
    
    // 检查是否有<i>或<em>标签的子元素 - 添加斜体
    if (cell.querySelector('i, em') || cell.innerHTML.includes('<i>') || cell.innerHTML.includes('<em>')) {
      style.font.italic = true;
    }
    
    // 检查是否有<u>标签的子元素 - 添加下划线
    if (cell.querySelector('u') || cell.innerHTML.includes('<u>')) {
      style.font.underline = true;
    }
    
    // 检查是否有<center>标签或style="text-align:center"
    if (cell.querySelector('center') || 
        cell.innerHTML.includes('<center>') || 
        cell.style.textAlign === 'center' || 
        cell.getAttribute('align') === 'center') {
      style.alignment.horizontal = 'center';
    } else if (cell.style.textAlign === 'right' || cell.getAttribute('align') === 'right') {
      style.alignment.horizontal = 'right';
    } else if (cell.style.textAlign === 'left' || cell.getAttribute('align') === 'left') {
      style.alignment.horizontal = 'left';
    }
    
    // 检查是否有带有color属性的span标签或font标签
    const colorElements = cell.querySelectorAll('[style*="color"], font[color]');
    if (colorElements.length > 0) {
      colorElements.forEach(el => {
        const colorAttr = el.getAttribute('color');
        const styleColor = el.style && el.style.color;
        
        if (colorAttr) {
          style.font.color = { rgb: this._convertColorToHex(colorAttr) };
        } else if (styleColor) {
          style.font.color = { rgb: this._convertColorToHex(styleColor) };
        }
      });
    }
    
    // 检查是否有font标签带size属性
    const fontElements = cell.querySelectorAll('font[size]');
    if (fontElements.length > 0) {
      fontElements.forEach(el => {
        const sizeAttr = el.getAttribute('size');
        if (sizeAttr) {
          // HTML font size 1-7, 转换为磅
          const htmlSizes = { '1': 8, '2': 10, '3': 12, '4': 14, '5': 18, '6': 24, '7': 36 };
          if (htmlSizes[sizeAttr]) {
            style.font.sz = htmlSizes[sizeAttr];
          }
        }
      });
    }
  }
  
  /**
   * 提取边框样式
   * @private
   * @param {HTMLElement} cell - 单元格元素
   * @param {Object} style - 样式对象
   * @param {CSSStyleDeclaration} computedStyle - 计算后的样式
   */
  _extractBorderStyles(cell, style, computedStyle) {
    // 边框样式映射
    const borderStyleMap = {
      'none': 'none',
      'hidden': 'none',
      'solid': 'thin',
      'double': 'double',
      'dotted': 'dotted',
      'dashed': 'dashed',
      'groove': 'medium',
      'ridge': 'medium',
      'inset': 'thin',
      'outset': 'thin'
    };
    
    // 处理各个边
    const borders = ['top', 'right', 'bottom', 'left'];
    
    borders.forEach(side => {
      // 获取边框样式、宽度和颜色
      const borderStyle = cell.style[`border${side.charAt(0).toUpperCase() + side.slice(1)}Style`] || 
                         computedStyle[`border${side.charAt(0).toUpperCase() + side.slice(1)}Style`];
      
      const borderWidth = cell.style[`border${side.charAt(0).toUpperCase() + side.slice(1)}Width`] || 
                         computedStyle[`border${side.charAt(0).toUpperCase() + side.slice(1)}Width`];
      
      const borderColor = cell.style[`border${side.charAt(0).toUpperCase() + side.slice(1)}Color`] || 
                         computedStyle[`border${side.charAt(0).toUpperCase() + side.slice(1)}Color`];
      
      // 如果存在边框，添加到样式中
      if (borderStyle && borderStyle !== 'none' && borderStyle !== 'hidden') {
        let xlsxBorderStyle = 'thin'; // 默认细边框
        
        // 根据宽度确定边框粗细
        if (borderWidth) {
          const widthValue = parseFloat(borderWidth);
          if (widthValue <= 1) xlsxBorderStyle = 'thin';
          else if (widthValue <= 2) xlsxBorderStyle = 'medium';
          else xlsxBorderStyle = 'thick';
        }
        
        // 使用映射表获取Excel边框样式
        if (borderStyleMap[borderStyle]) {
          xlsxBorderStyle = borderStyleMap[borderStyle];
        }
        
        style.border[side] = {
          style: xlsxBorderStyle,
          color: { rgb: this._convertColorToHex(borderColor || '#000000') }
        };
      }
    });
    
    // 检查单元格是否有border属性
    const border = cell.style.border || computedStyle.border;
    if (border && border !== 'none' && border !== '') {
      // 提取边框颜色
      let borderColor = '#000000';
      const borderColorMatch = border.match(/(#[0-9A-Fa-f]{3,8}|rgb\([^)]+\)|rgba\([^)]+\))/);
      if (borderColorMatch) {
        borderColor = borderColorMatch[1];
      }
      
      // 提取边框宽度和样式
      let xlsxBorderStyle = 'thin';
      if (border.includes('thick') || border.includes('3px') || border.includes('4px')) {
        xlsxBorderStyle = 'thick';
      } else if (border.includes('medium') || border.includes('2px')) {
        xlsxBorderStyle = 'medium';
      }
      
      // 应用到所有边
      borders.forEach(side => {
        style.border[side] = {
          style: xlsxBorderStyle,
          color: { rgb: this._convertColorToHex(borderColor) }
        };
      });
    }
    
    // 检查是否有border属性或frame属性
    const borderAttr = cell.getAttribute('border');
    const frameAttr = cell.getAttribute('frame');
    
    if (borderAttr && parseInt(borderAttr, 10) > 0) {
      // 应用边框到所有边
      const borderStyle = parseInt(borderAttr, 10) > 1 ? 'medium' : 'thin';
      borders.forEach(side => {
        style.border[side] = {
          style: borderStyle,
          color: { rgb: '000000' }
        };
      });
    } else if (frameAttr) {
      // 根据frame属性设置边框
      switch (frameAttr.toLowerCase()) {
        case 'box':
        case 'border':
          borders.forEach(side => {
            style.border[side] = { style: 'thin', color: { rgb: '000000' } };
          });
          break;
        case 'above':
          style.border.top = { style: 'thin', color: { rgb: '000000' } };
          break;
        case 'below':
          style.border.bottom = { style: 'thin', color: { rgb: '000000' } };
          break;
        case 'hsides':
          style.border.top = { style: 'thin', color: { rgb: '000000' } };
          style.border.bottom = { style: 'thin', color: { rgb: '000000' } };
          break;
        case 'vsides':
          style.border.left = { style: 'thin', color: { rgb: '000000' } };
          style.border.right = { style: 'thin', color: { rgb: '000000' } };
          break;
      }
    }
  }

  /**
   * 将CSS颜色值转换为十六进制格式
   * @private
   * @param {String} color - CSS颜色值
   * @returns {String} 十六进制颜色值（不带#号）
   */
  _convertColorToHex(color) {
    try {
      // 创建一个临时元素用于颜色转换
      const tempElement = document.createElement('div');
      tempElement.style.color = color;
      document.body.appendChild(tempElement);
      
      // 获取计算后的颜色
      const computedColor = window.getComputedStyle(tempElement).color;
      document.body.removeChild(tempElement);
      
      // 解析RGB值
      const rgbMatch = computedColor.match(/rgba?\((\d+),\s*(\d+),\s*(\d+)(?:,\s*[\d.]+)?\)/);
      if (rgbMatch) {
        const r = parseInt(rgbMatch[1], 10);
        const g = parseInt(rgbMatch[2], 10);
        const b = parseInt(rgbMatch[3], 10);
        
        // 转换为十六进制并去掉#号
        return ((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1).toUpperCase();
      }
    } catch (error) {
      if (this.options.debug) {
        console.warn('颜色转换出错:', error);
      }
    }
    
    return '000000'; // 默认黑色
  }

  /**
   * 将处理好的数据导出为Excel文件
   * @private
   * @param {Array} data - 表格数据
   * @param {Array} merges - 合并单元格信息
   * @param {Array} styles - 样式信息
   * @param {String} fileName - 文件名
   * @param {Object} sheetOptions - 工作表选项
   * @returns {Promise} 导出结果的Promise
   */
  _exportToExcel(data, merges = [], styles = [], fileName = '导出表格', sheetOptions = {}) {
    return new Promise((resolve, reject) => {
      try {
        // 创建工作簿
        const wb = XLSX.utils.book_new();
        
        // 将复杂数据转换为简单数组格式（仅包含值）
        const flatData = Array.isArray(data.data) ? data.data.map(row => 
          row.map(cell => (cell && typeof cell === 'object') ? cell.value : cell)
        ) : data.map(row => 
          row.map(cell => (cell && typeof cell === 'object') ? cell.value : cell)
        );
        
        // 创建工作表
        const ws = XLSX.utils.aoa_to_sheet(flatData);
        
        // 如果有合并单元格，添加到工作表
        if (merges && merges.length > 0) {
          ws['!merges'] = merges;
        }
        
        // 应用自定义列宽（如果提供）
        if (sheetOptions.colWidths) {
          ws['!cols'] = sheetOptions.colWidths.map(width => ({ wch: width }));
        } else {
          // 自动调整列宽
          this._autoAdjustColumnWidths(ws, flatData);
        }
        
        // 应用自定义行高（如果提供）
        if (sheetOptions.rowHeights) {
          ws['!rows'] = sheetOptions.rowHeights.map(height => ({ hpt: height }));
        } else {
          // 自动调整行高
          this._autoAdjustRowHeights(ws, flatData);
        }
        
        // 应用样式
        this._addStylesInfo(ws, flatData);
        
        // 应用单元格样式
        this._applyCellStyles(ws, data, styles);
        
        // 最重要：强制确保所有单元格居中对齐
        this._ensureCenterAlignment(ws);
        
        // 将工作表添加到工作簿
        XLSX.utils.book_append_sheet(wb, ws, 'Sheet1');
        
        // 导出为Excel文件
        XLSX.writeFile(wb, `${fileName}.xlsx`);
        
        resolve({
          success: true,
          fileName: `${fileName}.xlsx`
        });
      } catch (error) {
        if (this.options.debug) {
          console.error('导出Excel时出错:', error);
        }
        reject(error);
      }
    });
  }

  /**
   * 自动调整工作表的列宽
   * @private
   * @param {Object} ws - 工作表对象
   * @param {Array} data - 表格数据
   */
  _autoAdjustColumnWidths(ws, data) {
    // 初始化列宽数组
    const colWidths = [];
    
    // 计算每列的最大宽度
    data.forEach(row => {
      row.forEach((cell, colIndex) => {
        const content = cell ? String(cell) : '';
        
        // 根据内容长度估算宽度
        let width = this._estimateTextWidth(content);
        
        // 更新列的最大宽度
        colWidths[colIndex] = Math.max(colWidths[colIndex] || 0, width);
      });
    });
    
    // 应用列宽到工作表
    const cols = [];
    colWidths.forEach((width) => {
      // 设置最小宽度，确保不会太窄
      const adjustedWidth = Math.max(width, 8);
      
      // 限制最大宽度
      const finalWidth = Math.min(adjustedWidth, 100);
      
      cols.push({ wch: finalWidth });
    });
    
    ws['!cols'] = cols;
  }

  /**
   * 估算文本在Excel中的宽度
   * @private
   * @param {String} text - 文本内容
   * @returns {Number} - 估算的宽度值
   */
  _estimateTextWidth(text) {
    if (!text) return 0;
    
    const str = String(text);
    
    // 中文字符和特殊符号占用更多宽度
    const wideChars = str.replace(/[\x00-\xff]/g, '').length; // 非ASCII字符
    const narrowChars = str.length - wideChars; // ASCII字符
    
    // 根据字符类型估算宽度
    return narrowChars + (wideChars * 2) + 2; // 加2作为额外缓冲
  }

  /**
   * 自动调整工作表的行高
   * @private
   * @param {Object} ws - 工作表对象
   * @param {Array} data - 表格数据
   */
  _autoAdjustRowHeights(ws, data) {
    // 创建行高数组
    const rowHeights = [];
    
    // 计算每行的适当高度
    data.forEach((row, rowIndex) => {
      let maxLinesInRow = 1;
      
      row.forEach(cell => {
        if (!cell) return;
        
        const content = String(cell);
        
        // 检测文本中的换行符，估算行数
        const lines = content.split(/\r\n|\r|\n/).length;
        
        // 检测长文本，可能需要自动换行
        const textLength = content.length;
        const estimatedLines = Math.ceil(textLength / 50); // 假设每50个字符可能需要换行
        
        // 取行数和估算行数的较大值
        const totalLines = Math.max(lines, estimatedLines);
        
        // 更新当前行的最大行数
        maxLinesInRow = Math.max(maxLinesInRow, totalLines);
      });
      
      // 计算行高，每行至少25磅
      const rowHeight = Math.max(maxLinesInRow * 18, 25);
      
      rowHeights[rowIndex] = { hpt: rowHeight };
    });
    
    // 应用行高到工作表
    ws['!rows'] = rowHeights;
  }

  /**
   * 添加样式信息到工作表
   * @private
   * @param {Object} ws - 工作表对象
   * @param {Array} data - 表格数据
   */
  _addStylesInfo(ws, data) {
    // 获取工作表范围
    const range = XLSX.utils.decode_range(ws['!ref']);
    
    // 为每个单元格设置通用属性
    for (let row = range.s.r; row <= range.e.r; ++row) {
      for (let col = range.s.c; col <= range.e.c; ++col) {
        const cellAddress = XLSX.utils.encode_cell({ r: row, c: col });
        const cell = ws[cellAddress];
        
        if (!cell) continue;
        
        // 初始化样式对象
        if (!cell.s) cell.s = {};
        
        // 设置默认字体属性
        cell.s.font = cell.s.font || {};
        if (this.options.defaultFontBold && !('bold' in cell.s.font)) {
          cell.s.font.bold = true;
        }
        if (!cell.s.font.sz) {
          cell.s.font.sz = this.options.defaultFontSize;
        }
        if (!cell.s.font.color) {
          cell.s.font.color = { rgb: this.options.defaultFontColor };
        }
        
        // 设置默认对齐方式 - 强制居中
        cell.s.alignment = cell.s.alignment || {};
        cell.s.alignment.horizontal = 'center';
        cell.s.alignment.vertical = 'center';
        cell.s.alignment.wrapText = true;
        
        // 设置默认边框
        cell.s.border = cell.s.border || {
          top: { style: 'thin', color: { rgb: '000000' } },
          bottom: { style: 'thin', color: { rgb: '000000' } },
          left: { style: 'thin', color: { rgb: '000000' } },
          right: { style: 'thin', color: { rgb: '000000' } }
        };
        
        // 如果是表头行，应用表头样式
        if (this.options.headerStyle && row === range.s.r) {
          cell.s.font.bold = true;
          cell.s.font.sz = (cell.s.font.sz || this.options.defaultFontSize) + 1;
          
          // 为表头添加底部边框加粗
          cell.s.border.bottom = {
            style: 'medium',
            color: { rgb: '000000' }
          };
          
          // 添加轻微的背景色以突出显示表头
          cell.s.fill = cell.s.fill || {
            patternType: 'solid',
            fgColor: { rgb: 'F2F2F2' } // 浅灰色
          };
        }
        
        // 应用斑马纹
        if (this.options.zebra && row % 2 === 1 && row !== range.s.r) {
          cell.s.fill = cell.s.fill || {
            patternType: 'solid',
            fgColor: { rgb: 'F9F9F9' } // 更浅的灰色
          };
        }
      }
    }
  }

  /**
   * 应用单元格样式到工作表
   * @private
   * @param {Object} ws - 工作表对象
   * @param {Array} data - 表格数据
   * @param {Array} styles - 样式信息数组
   */
  _applyCellStyles(ws, data, styles) {
    // 如果没有样式信息，直接返回
    if (!styles || styles.length === 0) return;
    
    // 遍历样式信息
    styles.forEach(styleInfo => {
      const { row, col, style } = styleInfo;
      const cellAddress = XLSX.utils.encode_cell({ r: row, c: col });
      
      // 确保单元格对象存在
      if (!ws[cellAddress]) {
        // 创建单元格如果不存在
        const cellValue = this._getValueAt(data, row, col);
        ws[cellAddress] = { v: cellValue || '', t: 's' };
      }
      
      // 初始化样式对象如果不存在
      if (!ws[cellAddress].s) {
        ws[cellAddress].s = {};
      }
      
      // 深度合并样式对象
      ws[cellAddress].s = this._deepMergeStyles(ws[cellAddress].s, style);
    });
  }

  /**
   * 深度合并样式对象
   * @private
   * @param {Object} target - 目标样式对象
   * @param {Object} source - 源样式对象
   * @returns {Object} 合并后的样式对象
   */
  _deepMergeStyles(target, source) {
    const output = Object.assign({}, target);
    
    if (this._isObject(target) && this._isObject(source)) {
      Object.keys(source).forEach(key => {
        if (this._isObject(source[key])) {
          if (!(key in target)) {
            Object.assign(output, { [key]: source[key] });
          } else {
            output[key] = this._deepMergeStyles(target[key], source[key]);
          }
        } else {
          Object.assign(output, { [key]: source[key] });
        }
      });
    }
    
    return output;
  }

  /**
   * 确保所有单元格都有居中对齐设置
   * @private
   * @param {Object} ws - 工作表对象
   */
  _ensureCenterAlignment(ws) {
    // 获取工作表范围
    const range = XLSX.utils.decode_range(ws['!ref']);
    
    // 遍历所有单元格
    for (let row = range.s.r; row <= range.e.r; ++row) {
      for (let col = range.s.c; col <= range.e.c; ++col) {
        const cellAddress = XLSX.utils.encode_cell({ r: row, c: col });
        const cell = ws[cellAddress];
        
        if (!cell) continue;
        
        // 确保有样式对象
        if (!cell.s) {
          cell.s = {};
        }
        
        // 确保有对齐属性
        if (!cell.s.alignment) {
          cell.s.alignment = {};
        }
        
        // 强制水平居中
        cell.s.alignment.horizontal = 'center';
        
        // 强制垂直居中
        cell.s.alignment.vertical = 'center';
        
        // 启用自动换行
        cell.s.alignment.wrapText = true;
        
        // 确保每个单元格都有边框
        if (!cell.s.border) {
          cell.s.border = {
            top: { style: 'thin', color: { rgb: '000000' } },
            bottom: { style: 'thin', color: { rgb: '000000' } },
            left: { style: 'thin', color: { rgb: '000000' } },
            right: { style: 'thin', color: { rgb: '000000' } }
          };
        }
        
        // 确保有字体样式
        if (!cell.s.font) {
          cell.s.font = {
            name: 'Arial',
            sz: 11,
            color: { rgb: '000000' }
          };
        }
      }
    }
  }

  /**
   * 从数据数组中获取指定位置的值
   * @private
   * @param {Array} data - 数据数组
   * @param {Number} row - 行索引
   * @param {Number} col - 列索引
   * @returns {*} 单元格值
   */
  _getValueAt(data, row, col) {
    try {
      if (Array.isArray(data)) {
        if (row >= 0 && row < data.length && col >= 0) {
          const rowData = data[row];
          if (Array.isArray(rowData) && col < rowData.length) {
            const cell = rowData[col];
            return (cell && typeof cell === 'object') ? cell.value : cell;
          }
        }
      } else if (data && data.data && Array.isArray(data.data)) {
        // 处理 {data: [...]} 格式
        return this._getValueAt(data.data, row, col);
      }
    } catch (error) {
      if (this.options.debug) {
        console.warn(`获取单元格值时出错 [${row},${col}]:`, error);
      }
    }
    return '';
  }

  /**
   * 检查是否是对象
   * @private
   * @param {*} item - 检查的项目
   * @returns {Boolean} 是否是对象
   */
  _isObject(item) {
    return (item && typeof item === 'object' && !Array.isArray(item));
  }
  
  /**
   * 自动检测页面上的表格并导出
   * @param {String} fileName - 导出的文件名
   * @returns {Promise} 导出结果的Promise
   */
  autoDetectAndExportTables(fileName = this.options.fileName) {
    return new Promise((resolve, reject) => {
      try {
        // 获取页面上所有表格
        const allTables = document.querySelectorAll('table');
        
        if (this.options.debug) {
          console.log(`页面上共有 ${allTables.length} 个表格`);
        }
        
        // 如果没有表格，直接返回错误
        if (allTables.length === 0) {
          reject(new Error('页面上没有找到表格'));
          return;
        }
        
        // 检查是否存在嵌套表格
        const nestedTableContainers = [];
        
        // 遍历所有表格，寻找包含其他表格的容器
        allTables.forEach(table => {
          const nestedTables = table.querySelectorAll('table');
          if (nestedTables.length > 0) {
            nestedTableContainers.push(table);
          }
        });
        
        // 如果找到嵌套表格，导出第一个
        if (nestedTableContainers.length > 0) {
          if (this.options.debug) {
            console.log(`找到 ${nestedTableContainers.length} 个嵌套表格容器，导出第一个`);
          }
          
          this.exportTableFromDOM(nestedTableContainers[0], fileName)
            .then(resolve)
            .catch(reject);
          return;
        }
        
        // 如果有多个表格，尝试分组
        if (allTables.length > 1) {
          // 查找彼此靠近的表格
          const tableGroups = this._groupTables(allTables);
          
          if (this.options.debug) {
            console.log(`将表格分组为 ${tableGroups.length} 组，导出第一组`);
          }
          
          // 创建一个临时容器用于第一组
          const container = document.createElement('div');
          
          // 将第一组中的表格复制到容器中
          tableGroups[0].forEach(table => {
            container.appendChild(table.cloneNode(true));
          });
          
          // 导出这个容器
          this.exportTableFromDOM(container, fileName)
            .then(resolve)
            .catch(reject);
          return;
        }
        
        // 如果只有一个表格，直接导出
        this.exportTableFromDOM(allTables[0], fileName)
          .then(resolve)
          .catch(reject);
      } catch (error) {
        reject(error);
      }
    });
  }
  
  /**
   * 将表格分组
   * @private
   * @param {NodeList} tables - 表格元素集合
   * @returns {Array} 表格分组数组
   */
  _groupTables(tables) {
    const tableGroups = [];
    let currentGroup = [tables[0]];
    
    for (let i = 1; i < tables.length; i++) {
      const prevTable = tables[i-1];
      const currTable = tables[i];
      
      // 计算两个表格之间的距离
      const prevRect = prevTable.getBoundingClientRect();
      const currRect = currTable.getBoundingClientRect();
      const distance = currRect.top - (prevRect.top + prevRect.height);
      
      // 如果距离小于阈值，认为它们是同一组
      if (distance < 50) { // 50像素作为阈值
        currentGroup.push(currTable);
      } else {
        tableGroups.push([...currentGroup]);
        currentGroup = [currTable];
      }
    }
    
    // 添加最后一组
    if (currentGroup.length > 0) {
      tableGroups.push(currentGroup);
    }
    
    return tableGroups;
  }
}

// 创建Vue插件
const TableExportPlugin = {
  install(Vue, options = {}) {
    // 创建一个全局的导出器实例
    const globalExporter = new TableExporter(options);
    
    // 添加全局方法
    Vue.prototype.$exportTable = function(selector, fileName) {
      return globalExporter.exportTableFromDOM(selector, fileName);
    };
    
    Vue.prototype.$exportCurrentTable = function(selector, fileName) {
      return globalExporter.exportVueTable(this, selector, fileName);
    };
    
    Vue.prototype.$exportData = function(data, fileName, sheetOptions) {
      return globalExporter.exportDataToExcel(data, fileName, sheetOptions);
    };
    
    Vue.prototype.$autoExportTables = function(fileName) {
      return globalExporter.autoDetectAndExportTables(fileName);
    };
  }
};

export { TableExporter as default, TableExportPlugin };
