// 示例: 处理带有HTML标签和样式的表格
<template>
  <div>
    <div class="table-container">
      <table class="styled-table" border="1">
        <thead>
          <tr>
            <th colspan="4" style="font-size:16px;color:#0066cc;text-align:center;">
              <b>公司销售季度报表</b>
            </th>
          </tr>
          <tr>
            <th style="background-color:#f0f0f0;font-weight:bold;">产品名称</th>
            <th style="background-color:#f0f0f0;font-weight:bold;">一季度</th>
            <th style="background-color:#f0f0f0;font-weight:bold;">二季度</th>
            <th style="background-color:#f0f0f0;font-weight:bold;">增长率</th>
          </tr>
        </thead>
        <tbody>
          <tr>
            <td><b>产品A</b></td>
            <td>¥120,000</td>
            <td>¥150,000</td>
            <td style="color:green;"><i>+25%</i></td>
          </tr>
          <tr>
            <td><b>产品B</b></td>
            <td>¥85,000</td>
            <td>¥78,000</td>
            <td style="color:red;"><i>-8.2%</i></td>
          </tr>
          <tr>
            <td><b>产品C</b></td>
            <td>¥205,000</td>
            <td>¥240,000</td>
            <td style="color:green;"><i>+17%</i></td>
          </tr>
          <tr>
            <td style="font-weight:bold;">总计</td>
            <td style="font-weight:bold;">¥410,000</td>
            <td style="font-weight:bold;">¥468,000</td>
            <td style="font-weight:bold;color:green;"><i>+14.1%</i></td>
          </tr>
        </tbody>
      </table>
    </div>
    
    <button @click="exportTable" class="export-button">导出带样式的表格</button>
  </div>
</template>

<script>
import TableExporter from '@/utils/tableExporter';

export default {
  methods: {
    exportTable() {
      // 创建导出器实例 - 强制居中对齐
      const exporter = new TableExporter({
        defaultAlignment: 'center',  // 强制居中对齐
        headerStyle: true,           // 表头样式
        borders: true,               // 显示边框
        fileName: '带样式的销售报表', // 默认文件名
        debug: true                  // 启用调试
      });
      
      // 导出当前表格 - 会强制所有单元格居中对齐
      exporter.exportTableFromDOM('.styled-table', '销售报表')
        .then(result => {
          console.log('导出成功:', result);
        })
        .catch(error => {
          console.error('导出失败:', error);
        });
    }
  }
};
</script>
