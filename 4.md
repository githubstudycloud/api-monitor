我理解您的问题，原始样式值没能成功获取。这是因为在标准的 XLSX 库中，样式信息的读取和保存可能不完整。让我尝试提供一个更准确的解决方案。

要解决样式原值无法获取的问题，我们需要使用特殊的方法直接从原始 XLS 文件中提取样式信息。这里有一个更精确的方法：

```javascript
import * as XLSX from 'xlsx';
import ExcelJS from 'exceljs';

async function convertXlsBlobToXlsxWithStyles(xlsBlob) {
  try {
    // 步骤1: 首先创建一个二进制字符串
    const binaryString = await new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = (e) => resolve(e.target.result);
      reader.onerror = (e) => reject(e);
      reader.readAsBinaryString(xlsBlob);
    });
    
    // 步骤2: 使用 XLSX 读取数据 - 打开所有样式选项
    const workbook = XLSX.read(binaryString, { 
      type: 'binary',
      cellStyles: true,
      cellNF: true,
      cellDates: true,
      cellFormula: true,
      sheetStubs: true,
      bookVBA: true,
      cellHTML: true
    });
    
    // 步骤3: 直接使用 XLSX 的 write 函数导出为 xlsx
    // 这种方式会尽量保留原始样式
    const xlsxContent = XLSX.write(workbook, {
      bookType: 'xlsx',
      type: 'binary',
      cellStyles: true,
      compression: true
    });
    
    // 步骤4: 转换为 ArrayBuffer
    const buffer = new ArrayBuffer(xlsxContent.length);
    const view = new Uint8Array(buffer);
    for (let i = 0; i < xlsxContent.length; i++) {
      view[i] = xlsxContent.charCodeAt(i) & 0xFF;
    }
    
    // 步骤5: 创建 xlsx blob
    const xlsxBlob = new Blob([buffer], {
      type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
    });
    
    return xlsxBlob;
  } catch (error) {
    console.error('转换失败:', error);
    throw error;
  }
}
```

这个方法使用的是直接转换的方式，而不是尝试提取和重新应用样式。如果上述方法仍无法满足您的需求，您可以尝试使用一个专门设计用于保留样式的库：xlsx-style 或 xlsx-js-style。

还有一种方法是使用服务器端库进行转换，因为服务器端库通常对样式处理更完整：

```javascript
// 将 XLS blob 发送到服务器进行转换
async function convertXlsBlobToXlsxUsingServer(xlsBlob) {
  try {
    // 创建一个 FormData 对象用于上传
    const formData = new FormData();
    formData.append('file', xlsBlob, 'file.xls');
    
    // 发送到您的服务器端转换接口
    const response = await fetch('/api/convert-xls-to-xlsx', {
      method: 'POST',
      body: formData
    });
    
    if (!response.ok) {
      throw new Error('服务器转换失败');
    }
    
    // 获取转换后的 XLSX blob
    const xlsxBlob = await response.blob();
    return xlsxBlob;
  } catch (error) {
    console.error('转换失败:', error);
    throw error;
  }
}
```

在服务器端，您可以使用更强大的库如 Apache POI (Java)、EPPlus (.NET) 或 PHPSpreadsheet (PHP) 进行更准确的转换。

如果您想在纯前端环境中解决这个问题，且上述方法都不奏效，您可能需要尝试一个组合方法 - 先从 XLS 提取数据，然后手动应用默认样式：

```javascript
import * as XLSX from 'xlsx';
import ExcelJS from 'exceljs';

async function convertXlsToXlsxWithDefaultStyles(xlsBlob) {
  try {
    // 读取 XLS 数据
    const arrayBuffer = await xlsBlob.arrayBuffer();
    const data = new Uint8Array(arrayBuffer);
    const workbook = XLSX.read(data, { type: 'array' });
    
    // 创建新的 ExcelJS 工作簿
    const excelJsWorkbook = new ExcelJS.Workbook();
    
    // 遍历所有工作表
    for (const sheetName of workbook.SheetNames) {
      const worksheet = workbook.Sheets[sheetName];
      const jsonData = XLSX.utils.sheet_to_json(worksheet, { header: 1 });
      
      // 创建新工作表
      const newWorksheet = excelJsWorkbook.addWorksheet(sheetName);
      
      // 添加数据
      jsonData.forEach(row => {
        newWorksheet.addRow(row);
      });
      
      // 应用默认样式 - 所有单元格都有边框
      newWorksheet.eachRow(row => {
        row.eachCell(cell => {
          // 边框
          cell.border = {
            top: { style: 'thin' },
            left: { style: 'thin' },
            bottom: { style: 'thin' },
            right: { style: 'thin' }
          };
          
          // 文本对齐
          cell.alignment = {
            vertical: 'middle',
            horizontal: 'center',
            wrapText: true
          };
          
          // 默认字体
          cell.font = {
            name: 'Arial',
            size: 11
          };
        });
      });
      
      // 设置列宽 (固定宽度或按内容自适应)
      newWorksheet.columns.forEach(column => {
        column.width = 15; // 或根据需要调整
      });
    }
    
    // 导出为 xlsx 格式
    const buffer = await excelJsWorkbook.xlsx.writeBuffer();
    
    // 创建 xlsx blob
    const xlsxBlob = new Blob([buffer], {
      type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
    });
    
    return xlsxBlob;
  } catch (error) {
    console.error('转换失败:', error);
    throw error;
  }
}
```

这种方法不会保留原始样式，但会应用一套合理的默认样式，确保表格有边框、文本居中对齐等基本格式。

您可以根据自己的需求选择最适合的方法，或者组合使用这些方法来获得最佳结果。
